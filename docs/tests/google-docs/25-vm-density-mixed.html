Step-by-step guide

Step 1 — Set parameters

TOTAL_VMS=10
WINDOWS_COUNT=$(( TOTAL_VMS * 7 / 10 ))
RHEL_COUNT=$(( TOTAL_VMS - WINDOWS_COUNT ))
echo "Windows: ${WINDOWS_COUNT}  RHEL: ${RHEL_COUNT}"
UUID="mixed-fleet-$(date +%s)"

Step 2 — Create project and RBAC

oc new-project burner-mixed-fleet
oc create serviceaccount kube-burner -n burner-mixed-fleet
oc apply -f - <<'EOF'
apiVersion: rbac.authorization.k8s.io/v1
kind: ClusterRole
metadata:
  name: kube-burner
rules:
  - apiGroups: [""]
    resources: [namespaces, pods, services, configmaps, secrets, nodes, events, serviceaccounts, persistentvolumeclaims]
    verbs: [get, list, watch, create, delete, update, patch]
  - apiGroups: [kubevirt.io]
    resources: [virtualmachines, virtualmachineinstances]
    verbs: [get, list, watch, create, delete, update, patch]
  - apiGroups: [cdi.kubevirt.io]
    resources: [datavolumes]
    verbs: [get, list, watch, create, delete, update, patch]
  - apiGroups: [batch]
    resources: [jobs]
    verbs: [get, list, watch, create, delete, update, patch]
---
apiVersion: rbac.authorization.k8s.io/v1
kind: ClusterRoleBinding
metadata:
  name: kube-burner
roleRef:
  apiGroup: rbac.authorization.k8s.io
  kind: ClusterRole
  name: kube-burner
subjects:
  - kind: ServiceAccount
    name: kube-burner
    namespace: burner-mixed-fleet
EOF

Step 3 — Write config file

cat > /tmp/mixed-fleet-config.yml << EOF
global:
  gc: false
  measurements:
    - name: vmiLatency

jobs:
  - name: create-windows-vms
    namespace: burner-mixed-fleet
    jobType: create
    jobIterations: 1
    namespacedIterations: false
    podWait: false
    waitWhenFinished: false
    maxWaitTimeout: 60m
    objects:
      - objectTemplate: /config/windows-vm-template.yml
        replicas: ${WINDOWS_COUNT}

  - name: create-rhel-vms
    namespace: burner-mixed-fleet
    jobType: create
    jobIterations: 1
    namespacedIterations: false
    podWait: false
    waitWhenFinished: true
    maxWaitTimeout: 60m
    objects:
      - objectTemplate: /config/rhel-vm-template.yml
        replicas: ${RHEL_COUNT}

  - name: delete-all-vms
    namespace: burner-mixed-fleet
    jobType: delete
    waitForDeletion: true
    objects:
      - apiVersion: kubevirt.io/v1
        kind: VirtualMachine
        labelSelector: {app: mixed-fleet}
EOF

Step 4 — Write VM templates

Write Windows template to /tmp/windows-vm-template.yml and RHEL template to /tmp/rhel-vm-template.yml.

See the full markdown guide at docs/tests/markdown/25-vm-density-mixed.md for the complete template YAML.

Step 5 — Create ConfigMap

oc create configmap mixed-fleet-config \
  --from-file=config.yml=/tmp/mixed-fleet-config.yml \
  --from-file=windows-vm-template.yml=/tmp/windows-vm-template.yml \
  --from-file=rhel-vm-template.yml=/tmp/rhel-vm-template.yml \
  -n burner-mixed-fleet

Step 6 — Launch Job and stream logs

oc get pod -n burner-mixed-fleet -w
oc logs -f job/kb-mixed-fleet -n burner-mixed-fleet

Step 7 — Demo narrative

When RHEL VMs boot first: "See how the RHEL VMs are already running? 90 seconds — that's your Linux workloads, already live. The Windows VMs are still coming up — that's normal. OpenShift isn't adding overhead."

When Windows VMs reach Running: "There's your first Windows VM. Not on VMware. Not on bare metal. On Kubernetes. Managed by the same platform as your containers, with the same RBAC, same monitoring."

Step 8 — Clean up

oc delete job kb-mixed-fleet -n burner-mixed-fleet 2>/dev/null || true
oc delete vm -A -l app=mixed-fleet 2>/dev/null || true
sleep 60
oc delete project burner-mixed-fleet
oc delete clusterrole kube-burner 2>/dev/null || true
oc delete clusterrolebinding kube-burner 2>/dev/null || true
