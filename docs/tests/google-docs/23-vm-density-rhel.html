Test 23: RHEL 9 VM Fleet Density — Enterprise Linux at Scale

⭐⭐⭐⭐ Advanced ⏱ 20–40 minutes 🐧 RHEL 9 OpenShift Virtualization required

What it does: Creates N RHEL 9 virtual machines simultaneously and measures KubeVirt boot latency for enterprise Linux workloads
VM Image: quay.io/containerdisks/rhel9:9.0 (public, no auth)

What is this test?

CirrOS boots in 5 seconds. Your customers are not running CirrOS. They are running RHEL — the same operating system that runs their databases, their middleware, their compliance-critical workloads.

This test replaces the CirrOS disk with a real Red Hat Enterprise Linux 9 container disk and asks the same question: how many enterprise Linux VMs can OpenShift Virtualization boot at the same time?

Step-by-step guide

Step 1 — Open the web terminal

oc whoami
oc get hyperconverged -A
oc get nodes -l node-role.kubernetes.io/worker --no-headers | wc -l

Step 2 — Set your parameters

VM_COUNT=6
UUID="rhel-fleet-$(date +%s)"

Step 3 — Create project and RBAC

oc new-project burner-rhel-fleet
oc create serviceaccount kube-burner -n burner-rhel-fleet
oc apply -f - <<'EOF'
apiVersion: rbac.authorization.k8s.io/v1
kind: ClusterRole
metadata:
  name: kube-burner
rules:
  - apiGroups: [""]
    resources: [namespaces, pods, services, configmaps, secrets, nodes, events, serviceaccounts]
    verbs: [get, list, watch, create, delete, update, patch]
  - apiGroups: [kubevirt.io]
    resources: [virtualmachines, virtualmachineinstances]
    verbs: [get, list, watch, create, delete, update, patch]
  - apiGroups: [batch]
    resources: [jobs]
    verbs: [get, list, watch, create, delete, update, patch]
---
apiVersion: rbac.authorization.k8s.io/v1
kind: ClusterRoleBinding
metadata:
  name: kube-burner
roleRef:
  apiGroup: rbac.authorization.k8s.io
  kind: ClusterRole
  name: kube-burner
subjects:
  - kind: ServiceAccount
    name: kube-burner
    namespace: burner-rhel-fleet
EOF

Step 4 — Write the config file

cat > /tmp/rhel-fleet-config.yml << EOF
global:
  gc: false
  measurements:
    - name: vmiLatency

jobs:
  - name: create-rhel-vms
    namespace: burner-rhel-fleet
    jobType: create
    jobIterations: 1
    namespacedIterations: false
    podWait: false
    waitWhenFinished: true
    maxWaitTimeout: 30m
    objects:
      - objectTemplate: /config/rhel-vm-template.yml
        replicas: ${VM_COUNT}

  - name: delete-rhel-vms
    namespace: burner-rhel-fleet
    jobType: delete
    waitForDeletion: true
    objects:
      - apiVersion: kubevirt.io/v1
        kind: VirtualMachine
        labelSelector: {kube-burner-job: create-rhel-vms}
EOF

Step 5 — Write the RHEL VM template

cat > /tmp/rhel-vm-template.yml << 'EOF'
apiVersion: kubevirt.io/v1
kind: VirtualMachine
metadata:
  name: rhel-vm-{{.Iteration}}-{{.Replica}}
  labels:
    app: rhel-fleet
    os: rhel9
    kube-burner-job: create-rhel-vms
spec:
  running: true
  template:
    metadata:
      labels:
        app: rhel-fleet
        os: rhel9
    spec:
      domain:
        cpu:
          cores: 1
        resources:
          requests:
            memory: 2Gi
        devices:
          disks:
            - name: rootdisk
              disk:
                bus: virtio
            - name: cloudinitdisk
              disk:
                bus: virtio
          interfaces:
            - name: default
              masquerade: {}
      networks:
        - name: default
          pod: {}
      volumes:
        - name: rootdisk
          containerDisk:
            image: quay.io/containerdisks/rhel9:9.0
        - name: cloudinitdisk
          cloudInitNoCloud:
            userData: |
              #cloud-config
              hostname: rhel-vm-{{.Iteration}}-{{.Replica}}
              user: cloud-user
              password: redhat123
              chpasswd:
                expire: false
EOF

Step 6 — Create ConfigMap and launch Job

oc create configmap rhel-fleet-config \
  --from-file=config.yml=/tmp/rhel-fleet-config.yml \
  --from-file=rhel-vm-template.yml=/tmp/rhel-vm-template.yml \
  -n burner-rhel-fleet
cat << JOBYAML | oc apply -f -
apiVersion: batch/v1
kind: Job
metadata:
  name: kb-rhel-fleet
  namespace: burner-rhel-fleet
spec:
  backoffLimit: 0
  template:
    spec:
      serviceAccountName: kube-burner
      restartPolicy: Never
      initContainers:
        - name: copy-config
          image: quay.io/kube-burner/kube-burner:v2.6.1
          command: [sh, -c, "cp /config-src/* /config/"]
          volumeMounts:
            - {name: config-src, mountPath: /config-src}
            - {name: workdir,    mountPath: /config}
      containers:
        - name: kube-burner
          image: quay.io/kube-burner/kube-burner:v2.6.1
          workingDir: /config
          command: [kube-burner, init, -c, /config/config.yml, --uuid=${UUID}]
          volumeMounts:
            - {name: workdir, mountPath: /config}
      volumes:
        - name: config-src
          configMap:
            name: rhel-fleet-config
        - name: workdir
          emptyDir: {}
JOBYAML

Step 7 — Stream logs

oc get pod -n burner-rhel-fleet -w
oc logs -f job/kb-rhel-fleet -n burner-rhel-fleet

Step 8 — Clean up

oc delete job kb-rhel-fleet -n burner-rhel-fleet 2>/dev/null || true
oc delete configmap rhel-fleet-config -n burner-rhel-fleet 2>/dev/null || true
oc delete vm -A -l app=rhel-fleet 2>/dev/null || true
oc delete project burner-rhel-fleet
oc delete clusterrole kube-burner 2>/dev/null || true
oc delete clusterrolebinding kube-burner 2>/dev/null || true
