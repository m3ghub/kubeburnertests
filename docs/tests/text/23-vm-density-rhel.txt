================================================================================
TEST 23: RHEL 9 VM Fleet Density — Enterprise Linux at Scale
================================================================================
Difficulty:  Advanced (4 stars)
Time:        20-40 minutes
What it does: Creates N RHEL 9 VMs simultaneously and measures KubeVirt boot
             latency for enterprise Linux workloads
VM Image:    quay.io/containerdisks/rhel9:9.0 (public, no auth required)
Requires:    OpenShift Virtualization, multi-node cluster, quay.io reachable
================================================================================

WHAT IS THIS TEST?
------------------
CirrOS boots in 5 seconds. Your customers are not running CirrOS. They are
running RHEL — the same OS that runs their databases and compliance-critical
workloads. This test replaces CirrOS with Red Hat Enterprise Linux 9 and asks:
how many enterprise Linux VMs can OpenShift Virtualization boot at once?

BEFORE YOU START — OPEN THESE TABS
-----------------------------------
Tab 1: Console web terminal
Tab 2: Virtualization → VirtualMachines (watch VM state transitions)
Tab 3: Observe → Metrics (query: kubevirt_vmi_phase_count{phase="Running"})
Tab 4: Observe → Dashboards → KubeVirt / Infrastructure Resources / Top Consumers

PRE-FLIGHT CHECK
----------------
Verify RHEL image is pullable:
  oc run rhel-pull-test --image=quay.io/containerdisks/rhel9:9.0 \
    --restart=Never --command -- true -n default 2>/dev/null || true
  sleep 30
  oc get pod rhel-pull-test -n default
  oc delete pod rhel-pull-test -n default 2>/dev/null || true

If pod shows Completed — you are ready.
If ImagePullBackOff — quay.io is blocked on this cluster.

STEP 1 — OPEN THE WEB TERMINAL
-------------------------------
  oc whoami
  oc get hyperconverged -A
  oc get nodes -l node-role.kubernetes.io/worker --no-headers | wc -l

STEP 2 — SET YOUR PARAMETERS
------------------------------
  VM_COUNT=6
  UUID="rhel-fleet-$(date +%s)"

STEP 3 — CREATE PROJECT AND RBAC
----------------------------------
  oc new-project burner-rhel-fleet
  oc create serviceaccount kube-burner -n burner-rhel-fleet

  oc apply -f - <<'EOF'
  apiVersion: rbac.authorization.k8s.io/v1
  kind: ClusterRole
  metadata:
    name: kube-burner
  rules:
    - apiGroups: [""]
      resources: [namespaces, pods, services, configmaps, secrets, nodes,
                  events, serviceaccounts]
      verbs: [get, list, watch, create, delete, update, patch]
    - apiGroups: [kubevirt.io]
      resources: [virtualmachines, virtualmachineinstances]
      verbs: [get, list, watch, create, delete, update, patch]
    - apiGroups: [batch]
      resources: [jobs]
      verbs: [get, list, watch, create, delete, update, patch]
  ---
  apiVersion: rbac.authorization.k8s.io/v1
  kind: ClusterRoleBinding
  metadata:
    name: kube-burner
  roleRef:
    apiGroup: rbac.authorization.k8s.io
    kind: ClusterRole
    name: kube-burner
  subjects:
    - kind: ServiceAccount
      name: kube-burner
      namespace: burner-rhel-fleet
  EOF

STEP 4 — WRITE THE CONFIG FILE
--------------------------------
  cat > /tmp/rhel-fleet-config.yml << EOF
  global:
    gc: false
    measurements:
      - name: vmiLatency
  jobs:
    - name: create-rhel-vms
      namespace: burner-rhel-fleet
      jobType: create
      jobIterations: 1
      namespacedIterations: false
      podWait: false
      waitWhenFinished: true
      maxWaitTimeout: 30m
      objects:
        - objectTemplate: /config/rhel-vm-template.yml
          replicas: ${VM_COUNT}
    - name: delete-rhel-vms
      namespace: burner-rhel-fleet
      jobType: delete
      waitForDeletion: true
      objects:
        - apiVersion: kubevirt.io/v1
          kind: VirtualMachine
          labelSelector: {kube-burner-job: create-rhel-vms}
  EOF

STEP 5 — WRITE THE RHEL VM TEMPLATE
-------------------------------------
  cat > /tmp/rhel-vm-template.yml << 'EOF'
  apiVersion: kubevirt.io/v1
  kind: VirtualMachine
  metadata:
    name: rhel-vm-{{.Iteration}}-{{.Replica}}
    labels:
      app: rhel-fleet
      os: rhel9
      kube-burner-job: create-rhel-vms
  spec:
    running: true
    template:
      metadata:
        labels:
          app: rhel-fleet
          os: rhel9
      spec:
        domain:
          cpu:
            cores: 1
          resources:
            requests:
              memory: 2Gi
          devices:
            disks:
              - name: rootdisk
                disk:
                  bus: virtio
              - name: cloudinitdisk
                disk:
                  bus: virtio
            interfaces:
              - name: default
                masquerade: {}
        networks:
          - name: default
            pod: {}
        volumes:
          - name: rootdisk
            containerDisk:
              image: quay.io/containerdisks/rhel9:9.0
          - name: cloudinitdisk
            cloudInitNoCloud:
              userData: |
                #cloud-config
                hostname: rhel-vm-{{.Iteration}}-{{.Replica}}
                user: cloud-user
                password: redhat123
                chpasswd:
                  expire: false
  EOF

STEP 6 — CREATE CONFIGMAP
---------------------------
  oc create configmap rhel-fleet-config \
    --from-file=config.yml=/tmp/rhel-fleet-config.yml \
    --from-file=rhel-vm-template.yml=/tmp/rhel-vm-template.yml \
    -n burner-rhel-fleet

STEP 7 — LAUNCH THE JOB
-------------------------
  cat << JOBYAML | oc apply -f -
  apiVersion: batch/v1
  kind: Job
  metadata:
    name: kb-rhel-fleet
    namespace: burner-rhel-fleet
  spec:
    backoffLimit: 0
    template:
      spec:
        serviceAccountName: kube-burner
        restartPolicy: Never
        initContainers:
          - name: copy-config
            image: quay.io/kube-burner/kube-burner:v2.6.1
            command: [sh, -c, "cp /config-src/* /config/"]
            volumeMounts:
              - {name: config-src, mountPath: /config-src}
              - {name: workdir,    mountPath: /config}
        containers:
          - name: kube-burner
            image: quay.io/kube-burner/kube-burner:v2.6.1
            workingDir: /config
            command: [kube-burner, init, -c, /config/config.yml, --uuid=${UUID}]
            volumeMounts:
              - {name: workdir, mountPath: /config}
        volumes:
          - name: config-src
            configMap:
              name: rhel-fleet-config
          - name: workdir
            emptyDir: {}
  JOBYAML

STEP 8 — STREAM LOGS
---------------------
  oc get pod -n burner-rhel-fleet -w
  oc logs -f job/kb-rhel-fleet -n burner-rhel-fleet

WHAT TO LOOK FOR IN THE LOGS
------------------------------
  INFO vmiLatency:
  INFO   P50:  72s
  INFO   P95:  98s
  INFO   P99:  118s   <- RHEL boots slower than CirrOS — this is normal
  INFO Finished execution. UUID: rhel-fleet-...

STEP 9 — CLEAN UP
-----------------
  oc delete job kb-rhel-fleet -n burner-rhel-fleet 2>/dev/null || true
  oc delete configmap rhel-fleet-config -n burner-rhel-fleet 2>/dev/null || true
  oc delete vm -A -l app=rhel-fleet 2>/dev/null || true
  oc delete project burner-rhel-fleet
  oc delete clusterrole kube-burner 2>/dev/null || true
  oc delete clusterrolebinding kube-burner 2>/dev/null || true

TROUBLESHOOTING
---------------
  ImagePullBackOff on RHEL image
    -> quay.io/containerdisks/rhel9:9.0 unreachable — check pull secrets

  VMs stuck in Scheduling
    -> Not enough RAM — reduce VM_COUNT (each VM needs 2 Gi)

  serviceaccount "kube-burner" not found
    -> Re-run Step 3 — create SA separately first

  P99 > 5 minutes
    -> QEMU startup bottleneck — check: oc adm top nodes
